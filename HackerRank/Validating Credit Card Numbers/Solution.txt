import re

n = int(input())

for i in range(n):
    card = input()
    digits = card.replace("-", "")

    if card[0] not in "456":
        print("Invalid")
    elif not digits.isdigit():
        print("Invalid")
    elif len(digits) != 16:
        print("Invalid")
    elif "-" in card and not re.fullmatch(r"\d{4}-\d{4}-\d{4}-\d{4}", card):
        print("Invalid")
    elif re.search(r"(\d)\1{3}", digits):
        print("Invalid")
    else:
        print("Valid")
